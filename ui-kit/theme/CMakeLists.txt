cmake_minimum_required(VERSION 3.16)
set(ORGANIZATION_NAME "sberos")

set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(QT_FEATURE_zstd OFF)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Qml Quick Svg Xml DBus)


set(PLUGIN_NAME SberThemePlugin)
set(PLUGIN_VERSION "${VERSION_MAJOR}.${VERSION_MINOR}.${VERSION_PATCH}")

file(GLOB SOURCES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} *.cpp)
file(GLOB HEADERS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} *.h)
file(GLOB IMAGES_SOURCES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} images/*.cpp)
file(GLOB IMAGES_HEADERS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} images/*.h)
file(GLOB_RECURSE GENERATED_SOURCES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} generated/*.cpp)
file(GLOB_RECURSE GENERATED_HEADERS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} generated/*.h)

qt6_add_qml_module(${PLUGIN_NAME}
  URI Sber.Theme
  VERSION ${PLUGIN_VERSION}
  OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/qml/Sber/Theme/
  PLUGIN_TARGET ${PLUGIN_NAME}
  SOURCES
    ${SOURCES}
    ${HEADERS}
    ${IMAGES_SOURCES}
    ${IMAGES_HEADERS}
    ${GENERATED_SOURCES}
    ${GENERATED_HEADERS}
  DEPENDENCIES
    Sber.Core
)
target_include_directories(${PLUGIN_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/generated>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/generated/colors>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/generated/colors/general>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/generated/colors/additional>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/generated/colors/os>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/images>
  $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/core>
  $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/cmake>
  $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/utils>
)
add_dependencies(${PLUGIN_NAME} SberCorePlugin)
set_target_properties(${PLUGIN_NAME} PROPERTIES INSTALL_RPATH "$ORIGIN/../Core")
target_link_directories(${PLUGIN_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/qml/Sber/Core>
  ${QT_INSTALL_QML}/Sber/Core/
)
target_link_libraries(${PLUGIN_NAME} PUBLIC
  libSberCorePlugin.so
)
target_link_libraries(${PLUGIN_NAME} PRIVATE
  Qt6::Core
  Qt6::Gui
  Qt6::Widgets
  Qt6::Qml
  Qt6::Quick
  Qt6::Svg
  Qt6::Xml
  Qt6::DBus
)

sosc_install_plugin(${PLUGIN_NAME} ${PLUGIN_VERSION})
